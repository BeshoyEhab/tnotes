# tn Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `tn` Python CLI (JSONL + fzf, stdlib-only) replacing `~/.local/bin/note`, per approved design spec.

**Architecture:** Single `tnotes` package: `store.py` owns JSONL persistence + filtering; `extract.py` derives open/run actions from typed entries and markdown-inline syntax; `actions.py` shells out to opener/shell/clipboard/editor; `cli.py` parses args, renders safe-preview-first output; `picker.py` wraps the external `fzf` binary with fallback to `tn list`. No daemons, no deps beyond stdlib + `fzf` binary.

**Tech Stack:** Python 3.12+ (dev env is 3.14), stdlib-only runtime + external `fzf` binary, `uv` for project management, `pytest` for tests, fish for shell integration.

**Spec:** `docs/superpowers/specs/2026-09-29-tnotes-design.md`

## Global Constraints

- Python 3.12+, stdlib-only runtime + external `fzf` binary (no third-party runtime deps).
- Managed with `uv`; installed as `tn` shim → `~/.local/bin/tn` symlink to repo.
- Store: `~/.local/share/tnotes/tnotes.jsonl` (override with `TNOTES_DIR` env var); one JSON object per line.
- Entry schema: `id` 6-char lowercase hex unique; `type` plain string (`url|cmd|note`, extensible); `title` str; `body` str unlimited length; `tags` list[str]; `scope` `global|local`; `cwd` resolved absolute path for local, null for global; `status` `active|done`; `show_on_open` bool; `created`/`updated` ISO datetimes.
- Defaults on add: `scope=local` unless `-g`; `cwd=resolved $PWD` for local; `status=active`; `show_on_open=true` unless `--no-show`.
- Atomic append via temp-file + rename; Ctrl-C safe; failed editor writes preserve temp path in error.
- Type is explicit at add time (`--url/--cmd/--note`, default note); no `!`/`\` prefixes; literal `!`/`\`/`C:\path` stay plain text.
- Picker `tn` defaults here-only (global active + local active for `$PWD`); `tn --all` for everywhere; row format `{icon} {title} [{tags}] {scope}`, icon 🔗 url / `$` cmd / 📝 note, scope `G`/`·`, done shows `✔`.
- Preview capped at 100 body lines with `… (+N lines, Enter for full)`; renderer `bat -p` if present else plain wrapped text; `enter` = full body in `$PAGER` (`less -R`); keys `enter` full, `o` open, `r` run (always confirms `run? [y/N]`, prints exit code, never auto-runs on preview), `c` copy, `e` edit, `q` quit.
- Autoshow: `tn show --startup -l` per-dir silent when empty; done + `show_on_open=false` never autoshow; manual `tn show` still shows hidden (not done unless `--done`).
- Env backends: `EDITOR`→`VISUAL`→`vi`; `BROWSER`; `PAGER`; clipboard `wl-copy`→`xclip`→`xsel`; opener `xdg-open`→`$BROWSER`; missing opener/clipboard prints content + path; missing `fzf` falls back to `tn list`.
- Old `~/.local/bin/note` + fish hooks (`note-autoshow.fish`, `completions/note.fish`) and data `~/.local/share/mynotes/` are NOT touched or deleted in this plan.
- TDD: failing test first for every behavior; `pytest` green before done; frequent commits.

```python
# src/tnotes/store.py
"""JSONL persistence. Store: $TNOTES_DIR/tnotes.jsonl or ~/.local/share/tnotes/tnotes.jsonl."""
import json
import os
import secrets
import tempfile
from datetime import datetime, timezone
from pathlib import Path

SCHEMA_VERSION = 1

def data_dir() -> Path:
    override = os.environ.get("TNOTES_DIR")
    if override:
        return Path(override).expanduser()
    return Path.home() / ".local" / "share" / "tnotes"

def store_path() -> Path:
    return data_dir() / "tnotes.jsonl"

def _now_iso() -> str:
    return datetime.now(timezone.utc).astimezone().isoformat(timespec="seconds")

def generate_id(existing: set[str]) -> str:
    while True:
        cand = secrets.token_hex(3)  # 6 lowercase hex chars
        if cand not in existing:
            return cand

def load_all() -> list[dict]:
    p = store_path()
    if not p.exists():
        return []
    entries = []
    for line in p.read_text(encoding="utf-8").splitlines():
        if line.strip():
            entries.append(json.loads(line))
    return entries

def save_all(entries: list[dict]) -> None:
    d = data_dir()
    d.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=str(d), prefix="tnotes-", suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            for e in entries:
                f.write(json.dumps(e, ensure_ascii=False) + "\n")
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, store_path())
    except BaseException:
        try:
            os.unlink(tmp)
        except OSError:
            pass
        raise

def add_entry(type: str, title: str, body: str, tags: list[str],
              scope: str, cwd: str | None, show_on_open: bool) -> dict:
    entries = load_all()
    now = _now_iso()
    entry = {
        "id": generate_id({e["id"] for e in entries}),
        "type": type, "title": title, "body": body, "tags": list(tags),
        "scope": scope, "cwd": cwd, "status": "active",
        "show_on_open": show_on_open, "created": now, "updated": now,
    }
    entries.append(entry)
    save_all(entries)
    return entry

def get_entry(entry_id: str) -> dict | None:
    for e in load_all():
        if e["id"] == entry_id or e["id"].startswith(entry_id):
            return e
    return None

def update_entry(entry_id: str, **fields) -> dict:
    entries = load_all()
    for e in entries:
        if e["id"] == entry_id or e["id"].startswith(entry_id):
            e.update(fields)
            e["updated"] = _now_iso()
            save_all(entries)
            return e
    raise KeyError(f"no entry {entry_id}")

def delete_entry(entry_id: str) -> bool:
    entries = load_all()
    kept = [e for e in entries if not (e["id"] == entry_id or e["id"].startswith(entry_id))]
    if len(kept) == len(entries):
        return False
    save_all(kept)
    return True

def visible_entries(cwd: str | None = None, scope: str = "here",
                    include_done: bool = False, include_hidden: bool = False,
                    all_dirs: bool = False) -> list[dict]:
    if cwd is None:
        cwd = os.path.realpath(os.getcwd())
    out = []
    for e in load_all():
        if e.get("status") == "done" and not include_done:
            continue
        if not e.get("show_on_open", True) and not include_hidden:
            continue
        if all_dirs:
            out.append(e)
            continue
        if scope == "global":
            if e.get("scope") == "global":
                out.append(e)
        elif scope == "local":
            if e.get("scope") == "local" and e.get("cwd") == cwd:
                out.append(e)
        else:  # here
            if e.get("scope") == "global":
                out.append(e)
            elif e.get("scope") == "local" and e.get("cwd") == cwd:
                out.append(e)
    return out
```

```python
# src/tnotes/extract.py
"""Derive open/run actions. Typed entries first, then markdown-inline syntax.
Literal !, backslash, ~ never create actions."""
import re

_LINK_RE = re.compile(r"\[([^\]]*)\]\((https?://[^)\s]+)\)")
_BARE_RE = re.compile(r"https?://[^\s)>\]`'\"]+")
_INLINE_CMD_RE = re.compile(r"`([^`\n]+)`")
_FENCED_RE = re.compile(r"```(?:sh|bash|shell)?[ \t]*\n(.*?)```", re.DOTALL)

def _strip_trailing_punct(url: str) -> str:
    return url.rstrip(".,;:!?)]}'\"")

def _extract_links(text: str) -> list[dict]:
    acts, seen = [], set()
    for m in _LINK_RE.finditer(text):
        url = _strip_trailing_punct(m.group(2))
        if url not in seen:
            seen.add(url)
            acts.append({"kind": "open", "target": url,
                         "label": m.group(1) or url})
    for m in _BARE_RE.finditer(text):
        url = _strip_trailing_punct(m.group(0))
        if url not in seen:
            seen.add(url)
            acts.append({"kind": "open", "target": url, "label": url})
    return acts

def _extract_inline_cmds(text: str) -> list[dict]:
    no_fence = _FENCED_RE.sub("", text)
    return [{"kind": "run", "target": m.group(1).strip(), "label": m.group(1).strip()}
            for m in _INLINE_CMD_RE.finditer(no_fence) if m.group(1).strip()]

def _extract_fenced(text: str) -> list[dict]:
    acts = []
    for m in _FENCED_RE.finditer(text):
        for line in m.group(1).splitlines():
            line = line.strip()
            if line and not line.startswith("#"):
                acts.append({"kind": "run", "target": line, "label": line})
    return acts

def extract_actions(entry: dict) -> list[dict]:
    etype, body, title = entry.get("type", "note"), entry.get("body", ""), entry.get("title", "")
    if etype == "url":
        url = body.strip().splitlines()[0].strip() if body.strip() else ""
        base = [{"kind": "open", "target": url, "label": title or url}] if url else []
        return base + [a for a in _extract_links(body) if a["target"] != url]
    if etype == "cmd":
        lines = [ln.strip() for ln in body.splitlines() if ln.strip()]
        return [{"kind": "run", "target": ln, "label": title or ln} for ln in lines]
    return _extract_links(body) + _extract_inline_cmds(body) + _extract_fenced(body)
```

```python
# src/tnotes/actions.py
"""Side-effect backends. Order: opener xdg-open->$BROWSER;
clipboard wl-copy->xclip->xsel; editor $EDITOR->$VISUAL->vi."""
import os
import shutil
import subprocess

def find_exe(names: list[str]) -> str | None:
    for n in names:
        p = shutil.which(n)
        if p:
            return p
    return None

def open_target(url: str) -> int:
    opener = find_exe(["xdg-open"])
    if opener:
        return subprocess.run([opener, url]).returncode
    browser = os.environ.get("BROWSER")
    if browser:
        return subprocess.run(browser.split() + [url]).returncode
    print(f"no opener (xdg-open/$BROWSER) — URL:\n{url}")
    return 1

def run_command(cmd: str, confirm: bool = True) -> int:
    if confirm:
        ans = input(f"run? [y/N] {cmd}\n> ").strip().lower()
        if ans not in ("y", "yes"):
            print("aborted.")
            return 130
    p = subprocess.run(cmd, shell=True)
    print(f"exit: {p.returncode}")
    return p.returncode

def copy_text(text: str) -> bool:
    if find_exe(["wl-copy"]):
        subprocess.run(["wl-copy"], input=text.encode(), check=False)
        return True
    if find_exe(["xclip"]):
        subprocess.run(["xclip", "-selection", "clipboard"], input=text.encode(), check=False)
        return True
    if find_exe(["xsel"]):
        subprocess.run(["xsel", "--clipboard", "--input"], input=text.encode(), check=False)
        return True
    print("no clipboard tool (wl-copy/xclip/xsel) — content:\n" + text)
    return False

def edit_in_editor(path) -> int:
    editor = os.environ.get("EDITOR") or os.environ.get("VISUAL") or "vi"
    return subprocess.run([editor, str(path)]).returncode
```

```python
import argparse
import os
import sys
import tempfile
from pathlib import Path
from . import store
from . import actions

def _split_tags(s: str | None) -> list[str]:
    return [t.strip() for t in s.split(",") if t.strip()] if s else []

def _read_stdin_if_piped() -> str | None:
    if not sys.stdin.isatty():
        data = sys.stdin.read()
        return data if data.strip() else None
    return None

def compose_in_editor(entry_type: str, initial: str = "") -> str | None:
    header = (f"# tn {entry_type} — lines starting with # are ignored.\n"
              f"# Save+quit to save, quit with empty file to discard.\n")
    tmpl = {"url": "# URL: <paste url here>\n", "cmd": "# CMD: <paste command here>\n"}.get(entry_type, "")
    fd, tmp = tempfile.mkstemp(prefix="tn-", suffix=".md")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            f.write(header + tmpl + (initial or ""))
        if actions.edit_in_editor(tmp) != 0:
            print(f"editor exited non-zero; draft kept at {tmp}")
            return None
        lines = [ln for ln in Path(tmp).read_text(encoding="utf-8").splitlines()
                 if not ln.lstrip().startswith("#")]
        text = "\n".join(lines).strip()
        return text or None
    finally:
        try:
            Path(tmp).unlink()
        except OSError:
            pass

def _resolve_scope(args) -> tuple[str, str | None]:
    if args.g:
        return "global", None
    return "local", os.path.realpath(os.getcwd())

def cmd_add(args) -> int:
    etype = "note"
    if args.url is not None:
        etype = "url"
    elif args.cmd is not None:
        etype = "cmd"
    elif args.note is not None:
        etype = "note"
    body = args.url if etype == "url" else args.cmd if etype == "cmd" else args.note
    if body is None and args.text:
        body = args.text
    if body is None:
        body = _read_stdin_if_piped()
    if args.m:
        template_key = {"url": "URL: ", "cmd": "CMD: "}.get(etype, "")
        drafted = compose_in_editor(etype, (template_key + (body or "")) if body else "")
        if not drafted:
            print("discarded (empty).")
            return 1
        body = drafted
    if not body or not body.strip():
        print("error: empty entry; nothing saved.", file=sys.stderr)
        return 1
    body = body.strip()
    scope, cwd = _resolve_scope(args)
    title = args.title or (body.splitlines()[0][:60] if etype != "note" else body[:60])
    e = store.add_entry(type=etype, title=title, body=body,
                        tags=_split_tags(args.tags), scope=scope, cwd=cwd,
                        show_on_open=not args.no_show)
    print(f"saved {e['id']}")
    return 0

def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="tn", description="Terminal notepad with actions")
    p.add_argument("text", nargs="?", help="quick-add note text")
    p.add_argument("--url", nargs="?", const=None, default="__unset__", help="add URL entry")
    p.add_argument("--cmd", nargs="?", const=None, default="__unset__", help="add cmd entry")

    p.add_argument("--tags", default=None, help="comma-separated")

    p.add_argument("-m", action="store_true", help="compose in $EDITOR")

    p.add_argument("--hidden", action="store_true")
    p.add_argument("--startup", action="store_true")
    p.add_argument("--preview", action="store_true")

        s = sub.add_parser(name)
        if name in ("search", "show", "open", "run", "edit", "rm",
                    "done", "undone", "hide", "unhide"):

            s.add_argument("--done", action="store_true")
            s.add_argument("--hidden", action="store_true")
            s.add_argument("--startup", action="store_true")
            s.add_argument("--preview", action="store_true")

    if args.sub == "path":
        print(store.store_path())
        return 0

            return 1

            if not new_body:
                print("discarded (empty).")
                return 1
            store.update_entry(e["id"], body=new_body,
                               title=new_body.splitlines()[0][:60])
            print(f"updated {e['id']}")
            return 0
        finally:
            try:
                Path(tmp).unlink()
            except OSError:
                pass

                             or args.note is not None or args.text is not None or args.m
                             or not sys.stdin.isatty()):

            return cmd_add(args)
        return cmd_add(args)

if __name__ == "__main__":
```

```python
import os
import shutil
import subprocess
import sys
import textwrap
from . import store
from .extract import extract_actions
from .picker import format_row

def _autoshow_scope(args) -> tuple[str, bool]:
    if getattr(args, "g", False):
        return "global", False
    if getattr(args, "l", False):
        return "local", False
    return "here", False

def _entries_for(args, cwd=None):
    scope = "here"
    if getattr(args, "g", False):
        scope = "global"
    elif getattr(args, "l", False):
        scope = "local"
    return store.visible_entries(
        cwd=cwd, scope=scope,
        include_done=getattr(args, "done", False),
        include_hidden=True,
        all_dirs=getattr(args, "all", False))

def cmd_list(args) -> int:
    for e in _entries_for(args):
        print(f"{format_row(e)} {e['id']}")
    return 0

def cmd_search(args) -> int:
    q = (args.arg or "").lower()
    for e in _entries_for(args):
        if q in (e.get("title", "") + "\n" + e.get("body", "")).lower():
            print(f"{format_row(e)} {e['id']}")
    return 0

ICONS = {"url": "🔗", "cmd": "$", "note": "📝"}

def render_preview(entry: dict, actions: list[dict], max_lines: int = 100) -> str:
    icon = ICONS.get(entry.get("type"), "📝")
    tags = f" [{','.join(entry.get('tags', []))}]" if entry.get("tags") else ""
    scope = "(G)" if entry.get("scope") == "global" else "(·)"
    done = "✔ " if entry.get("status") == "done" else ""
    date = (entry.get("created") or "")[:10]
    lines = [f"{done}{icon} {entry.get('title', '')}{tags} {scope} {date}".rstrip()]
    if entry.get("type") in ("url", "cmd"):
        lines.append(entry.get("body", "").splitlines()[0] if entry.get("body") else "")
    show = "yes" if entry.get("show_on_open", True) else "no"
    lines.append(f"cwd: {entry.get('cwd') or '-'} / status: {entry.get('status')} / show: {show}")
    lines.append("---")
    lines.append("Actions:")
    oi = ri = 0
    for a in actions:
        if a["kind"] == "open":
            oi += 1
            lines.append(f" [o{oi}] open {a['target']}")
        else:
            ri += 1
            lines.append(f" [r{ri}] run {a['target']}")
    if not actions:
        lines.append(" (none — plain note)")
    lines.append(" [c] copy  [e] edit  [enter] full view")
    lines.append("---")
    lines.append("Body (wrapped):")
    body_lines = entry.get("body", "").splitlines() or [""]
    width = shutil.get_terminal_size((80, 24)).columns
    wrapped: list[str] = []
    for bl in body_lines:
        wrapped.extend(textwrap.wrap(bl, width) or [""])
    if len(wrapped) > max_lines:
        lines.extend(wrapped[:max_lines])
        lines.append(f"… (+{len(wrapped) - max_lines} lines, Enter for full)")
    else:
        lines.extend(wrapped)
    return "\n".join(lines) + "\n"

def render_full(entry: dict, actions: list[dict]) -> str:
    return render_preview(entry, actions, max_lines=10_000).replace(
        "… (+", "FULL — no truncation (+").replace(", Enter for full)", ")")

def _page(text: str) -> None:
    pager = os.environ.get("PAGER", "less -R")
    if not sys.stdout.isatty():
        print(text, end="")
        return
    if shutil.which("bat"):
        try:
            subprocess.run(["bat", "-p"], input=text.encode(), check=False)
            return
        except OSError:
            pass
    try:
        subprocess.run(pager.split(), input=text.encode(), check=False)
    except OSError:
        print(text, end="")

def cmd_show(args) -> int:
    if args.arg:  # show <id> [--preview]
        e = store.get_entry(args.arg)
        if not e:
            print(f"no entry {args.arg}", file=sys.stderr)
            return 1
        if getattr(args, "preview", False):
            sys.stdout.write(render_preview(e, extract_actions(e)))
        else:
            _page(render_full(e, extract_actions(e)))
        return 0
    entries = _entries_for(args)  # show (autoshow/manual)
    if getattr(args, "startup", False):
        entries = [e for e in entries
                   if e.get("status") == "active" and e.get("show_on_open", True)]
        if not entries:
            return 0  # silent when empty
    for e in entries:
        sys.stdout.write(render_preview(e, extract_actions(e)) + "\n")
    return 0

def dispatch(args) -> int:
    if args.sub == "list":
        return cmd_list(args)
    if args.sub == "search":
        return cmd_search(args)
    if args.sub == "show":
        return cmd_show(args)
    # open/run + bare picker land in Task 6
    from . import cli_actions as _a
    return _a.dispatch(args)
```

```python
    scope = "G" if entry.get("scope") == "global" else "·"
    done = "✔ " if entry.get("status") == "done" else ""
    return f"{done}{icon} {entry.get('title', '')}{tags} {scope}".rstrip()

def build_rows(entries: list[dict]) -> list[str]:
    return [f"{format_row(e)}\t{e['id']}" for e in entries]

def run_picker(entries: list[dict]) -> tuple[str | None, str | None]:
    """Returns (key, entry_id). key in enter/o/r/c/e/q or None on abort/fallback."""
    if not entries:
        print("no notes here yet — tn \"text\" to add one.")
        return None, None
    if not has_fzf():
        for e in entries:
            print(f"{format_row(e)} {e['id']}")
        return None, None
    tn = os.path.realpath(shutil.which("tn") or "tn")
    proc = subprocess.run(
        ["fzf", "--delimiter=\t", "--with-nth=1",
         "--preview", f"{tn} show {{2}} --preview",
         "--preview-window=right:60%:wrap",
         "--expect=enter,o,r,c,e,q"],
        input="\n".join(build_rows(entries)).encode(),
        capture_output=True)
    lines = proc.stdout.decode().splitlines()
    if len(lines) < 2:
        return None, None
    key = lines[0].strip() or "enter"
    entry_id = lines[1].split("\t")[-1].strip()
    return key, entry_id
```

```python
# src/tnotes/cli_actions.py
"""Picker entry + open/run/copy/edit key handling."""
import sys
from . import store, actions
from .extract import extract_actions
from . import picker as P
from .cli_read import render_full, _page, _entries_for

def _nth(actions_list: list[dict], kind: str, n: int) -> dict | None:
    items = [a for a in actions_list if a["kind"] == kind]
    return items[n - 1] if 1 <= n <= len(items) else None

def cmd_open(args) -> int:
    e = store.get_entry(args.arg)
    if not e:
        print(f"no entry {args.arg}", file=sys.stderr)
        return 1
    n = int(getattr(args, "n", None) or getattr(args, "arg2", None) or 1) if hasattr(args, "n") else 1
    try:
        n = int(sys.argv[sys.argv.index(args.arg) + 1]) if args.arg in sys.argv and len(sys.argv) > sys.argv.index(args.arg) + 1 and sys.argv[sys.argv.index(args.arg) + 1].isdigit() else n
    except (ValueError, IndexError):
        pass
    target = _nth(extract_actions(e), "open", n)
    if not target:
        print(f"no open action #{n} on {e['id']}", file=sys.stderr)
        return 1
    return actions.open_target(target["target"])

def cmd_run(args) -> int:
    e = store.get_entry(args.arg)
    if not e:
        print(f"no entry {args.arg}", file=sys.stderr)
        return 1
    n = 1
    try:
        i = sys.argv.index(args.arg)
        if len(sys.argv) > i + 1 and sys.argv[i + 1].isdigit():
            n = int(sys.argv[i + 1])
    except ValueError:
        pass
    target = _nth(extract_actions(e), "run", n)
    if not target:
        print(f"no run action #{n} on {e['id']}", file=sys.stderr)
        return 1
    return actions.run_command(target["target"], confirm=True)

def _handle_picked(key: str | None, entry_id: str | None) -> int:
    if not entry_id:
        return 0 if key is None else 1
    e = store.get_entry(entry_id)
    if not e:
        return 1
    acts = extract_actions(e)
    if key in (None, "", "enter"):
        _page(render_full(e, acts))
        return 0
    if key == "o":
        first = next((a for a in acts if a["kind"] == "open"), None)
        return actions.open_target(first["target"]) if first else 1
    if key == "r":
        first = next((a for a in acts if a["kind"] == "run"), None)
        return actions.run_command(first["target"], confirm=True) if first else 1
    if key == "c":
        actions.copy_text(e["body"])
        return 0
    if key == "e":
        from .cli import main as _main
        return _main(["edit", entry_id])
    return 0  # q/quit

def dispatch(args) -> int:
    if args.sub == "open":
        return cmd_open(args)
    if args.sub == "run":
        return cmd_run(args)
    # bare tn picker: tn [--all] [-g|-l] [--done] [--hidden]
    entries = _entries_for(args, cwd=None)
    if not getattr(args, "hidden", False):
        pass  # picker shows hidden by default (include_hidden=True in _entries_for)
    key, eid = P.run_picker(entries)
    return _handle_picked(key, eid)
```
